def write(p,x):p.write_text(json.dumps(x,indent=2)+'\n')
def clip(v,f,eq):return clip_triangles(v,f,np.asarray(eq),np.arange(len(f)))[:2]
def zbox(lo,hi):return np.array([[0,0,1,-hi],[0,0,-1,lo]])
def wedge(a,b):
    a,b=np.deg2rad([a,b]);return np.array([[np.sin(a),-np.cos(a),0,0],[-np.sin(b),np.cos(b),0,0]])
def hull(v):
    v=np.unique(np.round(v,12),axis=0)
    if len(v)<4 or np.linalg.matrix_rank(v-v.mean(0),tol=1e-10)<3:return None
    h=ConvexHull(v);ids=np.unique(h.simplices);vv=v[ids];f=np.searchsorted(ids,h.simplices);n=np.cross(vv[f[:,1]]-vv[f[:,0]],vv[f[:,2]]-vv[f[:,0]]);flip=np.einsum('ij,ij->i',n,vv[f].mean(1)-vv.mean(0))<0;f[flip]=f[flip][:,[0,2,1]]
    return vv,f,h.volume

def convex_regions(projected):
    # GEOS uses mm. These are partition guides only, not replacement vertices.
    polys=[]
    for tri in projected:
        p=Polygon(tri)
        if p.area>1e-8:polys.append(shapely.set_precision(p,.001))
    raw=shapely.union_all(polys);parts=list(raw.geoms)if raw.geom_type=='MultiPolygon'else[raw]
    cleaned=[];removed=[]
    for p in parts:
        if p.area<.01:removed.append(dict(type='tiny_component',area_mm2=p.area));continue
        holes=[]
        for ring in p.interiors:
            area=Polygon(ring).area
            if area>=.1:holes.append(ring.coords)
            else:removed.append(dict(type='tiny_projection_hole',area_mm2=area))
        cleaned.append(Polygon(p.exterior.coords,holes))
    material=shapely.union_all(cleaned);guide=material.simplify(.10,preserve_topology=True).buffer(.102,join_style='mitre')
    triangles=list(shapely.constrained_delaunay_triangles(guide).geoms)
    # Merge adjacent *2D source-contour triangles* only when their union is convex.
    # Not re-hulling old 3D proxy components, and no recursive error subdivision.
    active={i:t for i,t in enumerate(triangles)};neighbors=defaultdict(dict);edges=defaultdict(list)
    for i,t in active.items():
        xy=np.round(np.asarray(t.exterior.coords)[:-1],8)
        for a,b in zip(xy,np.roll(xy,-1,axis=0)):
            key=tuple(sorted([tuple(a),tuple(b)]));edges[key].append(i)
    for edge,ids in edges.items():
        if len(ids)==2:
            i,j=ids;length=float(np.linalg.norm(np.array(edge[0])-edge[1]));neighbors[i][j]=length;neighbors[j][i]=length
    queue=[];serial=0
    def enqueue(i,j):
        nonlocal serial
        if i not in active or j not in active or j not in neighbors[i]:return
        h=shapely.convex_hull(shapely.union_all([active[i],active[j]]))
        if h.area-active[i].area-active[j].area<1e-7:
            serial+=1;heapq.heappush(queue,(-neighbors[i][j],serial,i,j,h))
    for i in active:
        for j in neighbors[i]:
            if i<j:enqueue(i,j)
    while queue:
        _,_,i,j,_=heapq.heappop(queue)
        if i not in active or j not in active or j not in neighbors[i]:continue
        h=shapely.convex_hull(shapely.union_all([active[i],active[j]]))
        if h.area-active[i].area-active[j].area>=1e-7:continue
        active[i]=h;del active[j]
        affected=set(neighbors[i])|set(neighbors[j]);affected.discard(i);affected.discard(j)
        for n in affected:
            length=neighbors[i].get(n,0)+neighbors[j].get(n,0);neighbors[i][n]=length;neighbors[n].pop(j,None);neighbors[n][i]=length
        neighbors[i].pop(j,None);neighbors.pop(j,None)
        for n in affected:enqueue(i,n)
    regions=sorted(active.values(),key=lambda p:(p.centroid.x,p.centroid.y))
    error=shapely.union_all(regions).symmetric_difference(guide).area
    return regions,dict(original_area_mm2=raw.area,material_area_mm2=material.area,guide_area_mm2=guide.area,original_holes=sum(len(p.interiors)for p in parts),guide_holes=sum(len(p.interiors)for p in(list(guide.geoms)if guide.geom_type=='MultiPolygon'else[guide])),removed_projection_features=removed,triangles=len(triangles),convex_regions=len(regions),partition_area_difference_mm2=error,uncovered_original_projection_area_mm2=raw.difference(guide).area)

def main():
    out=ROOT/'build/parts';out.mkdir(exist_ok=False);records=[];skipped=[];profile_records=[]

        vv,ff=clip(upper_v,upper_f,wedge(a,b))

    summary=dict(parts=records,piece_counts={n:sum(r['mesh']==n for r in records)for n in ['link5','gripper_flange']},total_pieces=len(records),vertices=sum(r['vertices']for r in records),faces=sum(r['faces']for r in records),minimum_face_normal_width_mm=min(r['minimum_face_normal_width_mm']for r in records),minimum_volume_mm3=min(r['volume_mm3']for r in records),skipped_zero_volume_source_patches=skipped,profile_partitions=profile_records,retained_local_blocks=retained,retained_count=len(retained),source_vertex_policy='Every generated 3D vertex lies on an original clipped triangle; frozen local vertices are unchanged.     write(ROOT/'evidence/construction.json',summary);print(json.dumps({k:v for k,v in summary.items()if k not in ['parts','profile_partitions']},indent=2))
